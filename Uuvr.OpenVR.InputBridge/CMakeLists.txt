cmake_minimum_required(VERSION 3.24)
project(UuvrOpenVRInputBridge LANGUAGES CXX)
add_library(UuvrOpenVRInputBridge SHARED InputBridge.cpp)
target_compile_features(UuvrOpenVRInputBridge PRIVATE cxx_std_20)
target_compile_options(UuvrOpenVRInputBridge PRIVATE /W4 /utf-8)
target_include_directories(UuvrOpenVRInputBridge PRIVATE ../vendor/openvr-official/headers)
target_link_libraries(UuvrOpenVRInputBridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/openvr-official/lib/win64/openvr_api.lib")
set_target_properties(UuvrOpenVRInputBridge PROPERTIES OUTPUT_NAME "Uuvr.OpenVR.InputBridge")
enable_testing()
add_executable(InputBridgeTests InputBridgeTests.cpp)
target_compile_features(InputBridgeTests PRIVATE cxx_std_20)
target_compile_options(InputBridgeTests PRIVATE /W4 /utf-8)
target_include_directories(InputBridgeTests PRIVATE ../vendor/openvr-official/headers)
add_test(NAME InputBridgeContracts COMMAND InputBridgeTests)
